Binary keyframe format used in Grim Fandango

File structure:
  - Header
  - List of nodes with key entries
    Each entry contains:
       CndKeyNode
       std::vector<KeyNodeEntry>(KeyNode.numEntries)

struct KeyHeader final
{
    int32_t magic;   // 0x4659454B - FYEK aka KeyF
    char name[40];
    uint32_t numJoints;
    uint32_t type; // privileged 3DO node types
    float    fps;
    uint32_t numNodes;
    uint32_t numJoints2;
    uint32_t unknownMaybeFlags;
    uint32_t numUsedMarkers;

    float markerFrames[8];
    int32_t markerTypes[8];
};
sizeof(KeyHeader) == 180

List<KeyNode>(KeyHeader.numNodes)

struct KeyNode final
{
    char meshName[32];
    uint32_t nodeNum;
    uint32_t numEntries;
    uint32_t unknown;
};
sizeof(KeyNode) == 44

std::vector<KeyNodeEntry>(KeyNode.numEntries)

struct KeyNodeEntry // Same as in CND file from Infernal Machine
{
    float frame;
    int change_flag; // 0x1 - position changed, 0x2 - orientation changed
    Vector3f position;
    Vector3f orientation; // pyr
    Vector3f deltaPosition;
    Vector3f deltaOrientation; // pyr
};
sizeof(KeyNodeEntry) == 56
